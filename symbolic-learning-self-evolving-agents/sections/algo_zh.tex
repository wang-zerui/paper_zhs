\begin{algorithm}[t]
\caption{智能体符号学习框架}
\label{algo:main}
\begin{algorithmic}[1]
\Require $\Input$ \Comment{智能体系统的输入}
\Require $\Pipeline$ \Comment{包含节点的智能体流水线}
\Require $\bp$ \Comment{基于提示的梯度传播函数}
\Require $\Loss$ \Comment{基于提示的损失函数}
\Ensure 智能体系统中更新后的符号组件

\State $\Trajectory \gets []$ \Comment{初始化轨迹}

\State \textbf{前向传播}
\For{每个 $\Node \in \Pipeline$}
    \State $\NodeInput \gets \text{获取 } \Node \text{ 的输入}$ \Comment{节点输入}
    \State $\NodeOutput \gets \Node(\NodeInput, \NodePrompt, \NodeTool)$ \Comment{节点输出}
    \State 将 $(\NodeInput, \NodeOutput, \NodePrompt, \NodeTool)$ 追加到 $\Trajectory$
\EndFor

\State \textbf{损失计算}
\State $\LanguageLoss \gets \Loss(\Trajectory)$ \Comment{计算语言损失}

\State \textbf{反向传播}
\For{每个 $\Node \in \text{reverse}(\Pipeline)$}
    \State $\NodeGradient \gets \bp(\PreviousNodeGradient, \NodeInput, \NodeOutput, \NodePrompt, \NodeTool, \LanguageLoss)$
    \Comment{对最后一个节点，$\PreviousNodeGradient = \emptyset$}
    \State 将 $\NodeGradient$ 追加到 $\Trajectory$
\EndFor

\State \textbf{权重更新}
\For{每个 $\Node \in \Pipeline$}
    \State 使用 $\NodeGradient$ \text{更新} $\NodePrompt, \NodeTool$ \Comment{更新提示和工具}
\EndFor
\State 使用 $\{\NodeGradient\}$ \text{更新} $\Pipeline$ \Comment{更新智能体流水线}

\State \Return $(\Pipeline, \mathcal{P}, \mathcal{T})$ \Comment{更新后的智能体系统}
\end{algorithmic}
\end{algorithm}
