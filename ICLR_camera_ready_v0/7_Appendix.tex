\section{Sequence Placement Algorithm}\label{appendix:placement}

\noindent\textbf{Problem formulation.} We formalize planning over $D$ devices for a global batch of $N$ sequences. Let each sequence $i \in \{1,\dots,N\}$ have length $h_i$. We choose a sharding degree $p_i \in \{1,2,4,\dots,p_{\max}\}$ and an assignment $y_{i,d} \in \{0,1\}$ indicating whether device $d$ holds a shard of sequence $i$, such that $\sum_{d=1}^D y_{i,d} = p_i$. For executability, we restrict sharded sequences ($p_i>1$) to a hierarchical family of admissible device groups: for each degree $p$, the groups form a partition of GPUs (disjoint and covering), and partitions are nested across degrees.

\noindent\textbf{Cost model.} On each device $d$, we define (i) \textbf{memory load} $M_d = \sum_{i=1}^N (h_i / p_i)\, y_{i,d}$ with a hard threshold $M_d \le M_{\text{th}}=\alpha M^\star$ (we use $\alpha=1.1$), where $M^\star$ is the average per-device token count; and (ii) \textbf{compute load} $C_d \propto \sum_{i=1}^N (h_i^2 / p_i)\, y_{i,d}$ as an attention-cost proxy. Sharding also introduces communication overhead; following the standard SeqAlltoAll model, the per-device traffic volume scales as $\mathrm{vol}(h,p)\propto h\cdot(p-1)/p^2$, optionally weighted by intra-/inter-node bandwidth. In Algorithm~\ref{alg:placement}, we use $\mathrm{SplitCost}(h,p)$ as a practical proxy for this overhead.

\noindent\textbf{Objective.} We solve a lexicographic optimization:
\begin{equation}
    \min_{\{p_i, y_{i,d}\}} \left( \max_{d} C_d,\; \sum_{i=1}^N \mathrm{Comm}(h_i, p_i) \right),
\end{equation}
where we primarily minimize the maximum compute imbalance and use the communication cost to break ties.

\noindent\textbf{FlexUlysses 3-Stage Scheduling Heuristic.} The detailed sequence placement algorithm is formalized in Algorithm~\ref{alg:placement}. Our algorithm incorporates a threshold multiplier ($1.1\times$) for memory constraint $M_{\text{th}}$ and searches over power-of-two candidate degrees.

\begin{algorithm}
    \caption{FlexUlysses Placement Algorithm}
    \label{alg:placement}
    \begin{algorithmic}[1]
    \Require sequence heights $\mathcal{H}=\{h_i\}_{i=1}^N$, number of devices $D$
    \Ensure placement $\mathcal{Y}=\{(i,d,\hat h_{i,d},p_i)\}$ and per-item split degree $\mathcal{P}=\{p_i\}$

    \State $M^\star \gets \frac{1}{D}\sum_{i=1}^N h_i$, \quad $M_{\mathrm{th}}\gets \alpha M^\star$ \Comment{$\alpha=1.1$}
    \State $p_{\max}\gets 2^{\lfloor \log_2 D \rfloor}$,\quad $\mathcal{P}_{\mathrm{cand}}\gets\{2^k:0\le k\le \log_2 p_{\max}\}$
    \State $(\mathcal{Y}, \mathbf{m}) \gets \Call{BaselineSplitPlace}{\mathcal{H},D,M^\star,M_{\mathrm{th}},\mathcal{P}_{\mathrm{cand}}}$
    \State $\mathbf{s}\gets$ item states induced by $\mathcal{Y}$ \Comment{$\mathbf{s}[i].p \!=\!$ current split degree,\; $\mathbf{s}[i].G \!=\!$ device group}
    \State $\mathcal{L}\gets \{2^k:1\le k\le \log_2 p_{\max}\}$ in decreasing order
    \ForAll{$p \in \mathcal{L}$}
    \While{$\exists$ merge candidate at split level $p$}
        \State $\mathcal{C}_p \gets \Call{FindMergeCandidates}{\mathcal{Y},\mathbf{s},p,M_{\mathrm{th}},m}$ \Comment{$m=\texttt{DEFAULT\_M\_CANDIDATES}$}
        \If{$\mathcal{C}_p=\emptyset$} \State \textbf{break} \EndIf
        \State $c^\star \gets \arg\min_{c\in\mathcal{C}_p}\Delta\mathrm{Cost}(c)$
        \State $\Call{ExecuteMerge}{\mathcal{Y},\mathbf{s},c^\star}$
    \EndWhile
    \EndFor
    \State $\Call{ShortItemReorder}{\mathcal{Y},\mathbf{s},\mathcal{H},M_{\mathrm{th}}}$
    \State \Return $\mathcal{Y}, \mathcal{P}$
    \end{algorithmic}
\end{algorithm}

\vspace{0.5em}
\noindent\textbf{Split cost.}
\[
\mathrm{SplitCost}(h,p)=
\begin{cases}
0, & p\le 1\\
\frac{h(p-1)}{p^2}, & 2\le p\le \tau\\
\gamma\cdot\frac{h(p-1)}{p^2}, & p>\tau
\end{cases}
\quad\text{with }\tau=8,\ \gamma=16.
\]
\[
\mathrm{TotalCost}(\mathcal{P})=\sum_{i=1}^N \mathrm{SplitCost}(h_i,p_i).
\]
\begin{algorithm}
\caption{FlexUlysses Sub-procedures}
\label{alg:subprocedures}
\begin{algorithmic}[1]
\Procedure{BaselineSplitPlace}{$\{h_i\}_{i=1}^N, D, M^\star, M_{\mathrm{th}}, \mathcal{P}_{\mathrm{cand}}$}
\State $\mathbf{m}\gets \mathbf{0}$, \quad $\mathbf{a}\gets 1.1 M^\star\cdot \mathbf{1}$ \Comment{available memory proxy}
\State sort items by $h_i$ descending
\For{$i=1$ \textbf{to} $N$}
  \State $p \gets \Call{GetSPSize}{h_i,\mathcal{P}_{\mathrm{cand}}}$ \Comment{power-of-two}
  \While{$p \le p_{\max}$ \textbf{and} $\Call{ValidGroupExists}{p,i,\mathbf{m},M_{\mathrm{th}}}=\textbf{false}$}
    \State $p \gets 2p$
  \EndWhile
  \If{$\Call{ValidGroupExists}{p,i,\mathbf{m},M_{\mathrm{th}}}=\textbf{false}$} \State $p\gets p_{\max}$ \EndIf
  \State $G \gets \arg\max_{G:|G|=p,\ \forall d\in G: m_d+\frac{h_i}{p}\le M_{\mathrm{th}}}\ \sum_{d\in G} a_d$
  \ForAll{$d\in G$}
    \State add tuple $(i,d,\hat h_{i,d}\!=\!\frac{h_i}{p},p)$ to $\mathcal{Y}$
    \State $m_d \gets m_d + \frac{h_i}{p}$
    \State $a_d \gets a_d - \frac{h_i}{p}$ \Comment{proxy update}
  \EndFor
\EndFor
\State \Return $\mathcal{Y},\mathbf{m}$
\EndProcedure

\vspace{0.3em}
\Statex
\Procedure{FindMergeCandidates}{$\mathcal{Y}, \mathbf{s}, p, M_{\mathrm{th}}, m$}
\State partition devices into consecutive groups $\mathcal{G}_p=\{G\subseteq [D]: |G|=p\}$
\State $\mathcal{C}_p\gets \emptyset$
\ForAll{$G\in\mathcal{G}_p$}
  \State $I_G \gets \{i:\mathbf{s}[i].p=p \wedge \mathbf{s}[i].G=G\}$
  \If{$|I_G|<2$} \State \textbf{continue} \EndIf
  \For{$m' = 2$ \textbf{to} $m$}
    \State choose a multiset $S\subseteq I_G$ with $|S|=m'$ (greedy by increasing $h_i$)
    \State $(S_1,S_2)\gets \Call{TwoWayPack}{S}$ \Comment{subject to balance constraints}
    \If{$S_1\neq\emptyset \wedge S_2\neq\emptyset \wedge \Call{MergeFeasible}{G,S_1,S_2,p,M_{\mathrm{th}}}$}
      \State $\mathcal{C}_p \gets \mathcal{C}_p \cup \{c=(G,S_1,S_2,p)\}$
    \EndIf
  \EndFor
\EndFor
\State \Return $\mathcal{C}_p$
\EndProcedure

\vspace{0.3em}
\Statex
\Procedure{ExecuteMerge}{$\mathcal{Y},\mathbf{s},c=(G,S_1,S_2,p)$}
  \State $p' \gets p/2$,\quad split $G$ into halves $G^{(1)},G^{(2)}$ with $|G^{(1)}|=|G^{(2)}|=p'$
  \State remove all shards of items in $S_1\cup S_2$ from devices in $G$
  \ForAll{$d\in G^{(1)}$} \ForAll{$i\in S_1$} add $(i,d,h_i/p',p')$ to $\mathcal{Y}$ \EndFor \EndFor
  \ForAll{$d\in G^{(2)}$} \ForAll{$i\in S_2$} add $(i,d,h_i/p',p')$ to $\mathcal{Y}$ \EndFor \EndFor
  \ForAll{$i\in S_1$} $\mathbf{s}[i].p\gets p'$, $\mathbf{s}[i].G\gets G^{(1)}$ \EndFor
  \ForAll{$i\in S_2$} $\mathbf{s}[i].p\gets p'$, $\mathbf{s}[i].G\gets G^{(2)}$ \EndFor
\EndProcedure

\vspace{0.3em}
\Statex
\Procedure{ShortItemReorder}{$\mathcal{Y}, \mathbf{s}, \{h_i\}, M_{\mathrm{th}}$}
\State $S \gets \{i:\mathbf{s}[i].p\le 2\}$ sorted by decreasing $h_i$
\ForAll{$i\in S$}
  \State remove all tuples in $\mathcal{Y}$ with item $i$
  \State $\mathbf{s}[i].p\gets 1$, $\mathbf{s}[i].G\gets \emptyset$
\EndFor
\ForAll{$i\in S$}
  \State $d^\star \gets \arg\min_{d: m_d+h_i\le M_{\mathrm{th}}}\ \Big(\sum_{(j,d,\cdot,\cdot)\in\mathcal{Y}} h_j\cdot h_j\Big)$
  \If{no feasible $d^\star$} \State $d^\star \gets \arg\min_d (m_d+h_i)$ \EndIf
  \State add $(i,d^\star,h_i,1)$ to $\mathcal{Y}$; update $m_{d^\star}\gets m_{d^\star}+h_i$
\EndFor
\EndProcedure
\end{algorithmic}
\end{algorithm}
